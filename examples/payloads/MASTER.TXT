REM ==========================================================
REM  DOLOS MASTER TEST PAYLOAD
REM
REM  One payload that exercises the whole engine. Every check
REM  prints what it EXPECTED and what it GOT, so a failure is
REM  visible in the text itself - no need to know the answers.
REM
REM  Safe by construction: it types only into whatever text
REM  field has focus. No GUI key, no launcher, no CTRL/ALT
REM  chord that any editor treats as a command, no mouse
REM  clicks, no file or system access.
REM
REM  Open a plain text editor, click into it, then fire.
REM ==========================================================
DEFAULTDELAY 12
DELAY 1500
STRINGLN ################ DOLOS MASTER TEST ################

REM ==========================================================
REM  SECTION A - HOST AND ENVIRONMENT
REM ==========================================================
ENTER
STRINGLN --- A. HOST AND ENVIRONMENT ---
STRINGLN detected OS code (0=win 1=linux 2=mac) -> $_OS
STRINGLN caps lock now   -> $_CAPSLOCK_ON
STRINGLN num lock now    -> $_NUMLOCK_ON
STRINGLN scroll lock now -> $_SCROLLLOCK_ON
IF ($_OS == 2) THEN
STRINGLN host reads as MAC - accented text below should be correct
ELSE IF ($_OS == 1) THEN
STRINGLN host reads as LINUX
ELSE
STRINGLN host reads as WINDOWS
END_IF

STRINGLN precedence: * before +.    expect 22       got $c

STRINGLN parentheses override it.   expect 34       got $c
VAR $c = $a * $b + $a / $b
STRINGLN mixed precedence.          expect 62       got $c
